class Meniu{
private:
	int pret;
public:
	Meniu(int pret) : pret{pret} {}
	virtual int getPret() { return this->pret; }
	virtual string descriere() = 0;
	virtual ~Meniu() = default;
};

class CuRacoritoare : public Meniu{
private:
	Meniu* m;
public:
	CuRacoritoare(Meniu* m) : Meniu{ m.getPret() }, m{m} {}
	int getPret() override{
		return m->getPret() + 4;
	}

	string descriere() override{
		return m->descriere() + "cu racoritoare";
	}
	~CuRacoritoare() {
		delete m;
	}
}

class CuCafea : public Meniu{
private:
	Meniu* m;
public:
	CuCafea(Meniu* m) : Meniu{ m->getPret() }, m{m} {}
	int getPret() override{
		return m->getPret() + 5;	
	}
	
	string descriere() override{
		return m->descriere() + "cu cafea";
	}
	~CuCafea() {
		delete m;
	}
}

class MicDejun : public Meniu{
private:
	string denumire;
public:
	MicDejun(int pret, string denumire) : Meniu{pret}, denumire{denumire} {}
	string descriere() override {
		return this->denumire;
	}
}

template <typename T>
class Measurement{
private:
	T m;
public:
	Measurement(T m) : m{m} {}
	Measurement<T>& operator +(T v){
		this->m += v;
		return *this;
	}
	T value() const {
		return this->m;
	}
	bool operator <(const Measurement& m) const {
		return this->m < m.value();
	}
}
