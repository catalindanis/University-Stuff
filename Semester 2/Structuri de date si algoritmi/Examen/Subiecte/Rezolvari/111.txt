Subalgoritm sterge(cp, e)
	e<-cp.e[0]
	cp.e[0]<-cp.e[cp.n-1]
	cp.n<-cp.n-1
	coboara(cp, 0)
Sfarsit sterge

Subalgoritm coboara(cp, poz)
	e<-cp.e[poz]
	i<-poz
	j<-2*poz + 1
	Cat timp (j < cp.n) executa
		Daca (cp.R(cp.e[j+1], cp.e[j]) = true) atunci
			j<-j+1
		Sfarsit daca
		
		Daca(cp.R(e, cp.e[j]) = true) atunci
			j<-cp.n
		Altfel
			cp.e[i]<-cp.e[j]
			i<-j
			j<-2*i
		Sfarsit daca
	Sfarsit cat timp
	cp.e[i]<-e
Sfarsit coboara

Subalgoritm adauga(cp, e)
	cp.e[cp.n]<-e
	cp.n<-cp.n+1
	urca(cp, cp.n-1)
Sfarsit adauga
