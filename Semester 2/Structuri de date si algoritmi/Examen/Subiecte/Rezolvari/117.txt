Subalgoritm h(p)
{preconditii: p - nod
 postconditii: se returneaza inaltimea lui p}
	Daca p = NIL atunci
		h<--1
	Altfel
		h<-[p].h
	SfDaca
Sfarsit h

Subalgoritm inaltime(p)
{preconditii: p - nod
 postconditii: se recalculeaza si returneaza inaltimea lui p}
	Daca p = NIL atunci
		inaltime<--1
	Altfel
		inaltime<-max(inaltime([p].st), inaltime([p].dr)) + 1
	SfDaca
Sfarsit inaltime

Subalgoritm SRS(p)
{preconditii: p - radacina unui subarbore din arbore binar de cautare nevid
 postconditii: returneaza noua radacina a subarborelui in urma aplicarii srs}
	y<-[p].dr
	[p].dr<-[y].st
	[y].st<-p	
	[p].h<-inaltime(p)
	[y].h<-inaltime(y)
	SRS<-y
Sfarsit SRS

Subalgoritm DRD(p)
{preconditii: p - radacina unui subarbore din arbore binar de cautare nevid
 postconditii: returneaza noua radacina a subarborelui in urma aplicarii drd}
	Daca p.st != NIL atunci
		p<-SRS(p.st)
	Sfarsit daca
	DRD<-SRD(p)
